cmake_minimum_required(VERSION 3.0.2)
project(warehouse_recovery)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs geometry_msgs)
catkin_package(CATKIN_DEPENDS roscpp std_msgs geometry_msgs)
include_directories(${catkin_INCLUDE_DIRS})
add_executable(recovery_node src/recovery_node.cpp)
target_link_libraries(recovery_node ${catkin_LIBRARIES})
install(TARGETS recovery_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
